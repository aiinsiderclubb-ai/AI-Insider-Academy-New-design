-- Роль только для чтения для AI-агента.
-- Запустите от имени владельца базы (psql -d ВАША_БАЗА -f postgres-readonly-role.sql),
-- предварительно заменив пароль. Схема по умолчанию — public; для других схем повторите блок GRANT.

CREATE ROLE agent_readonly LOGIN PASSWORD 'ЗАМЕНИТЕ_НА_ДЛИННЫЙ_ПАРОЛЬ'
  NOSUPERUSER NOCREATEDB NOCREATEROLE NOINHERIT;

-- Каждый сеанс агента — только чтение и не дольше 15 секунд на запрос.
ALTER ROLE agent_readonly SET default_transaction_read_only = on;
ALTER ROLE agent_readonly SET statement_timeout = '15s';
ALTER ROLE agent_readonly SET idle_in_transaction_session_timeout = '60s';
ALTER ROLE agent_readonly CONNECTION LIMIT 5;

DO $$ BEGIN EXECUTE format('GRANT CONNECT ON DATABASE %I TO agent_readonly', current_database()); END $$;
GRANT USAGE ON SCHEMA public TO agent_readonly;
GRANT SELECT ON ALL TABLES IN SCHEMA public TO agent_readonly;
-- Таблицы, созданные позже владельцем, тоже будут доступны на чтение:
ALTER DEFAULT PRIVILEGES IN SCHEMA public GRANT SELECT ON TABLES TO agent_readonly;
